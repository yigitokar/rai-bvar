% Underscores may be written plain (cond_mean) or escaped (cond\_mean); \#, \%, \&, \$, \{ and \}
% are unescaped as well. Brackets, quotes, '::' and '=' print as typed. Works in text and in math

\ExplSyntaxOn
\str_new:N \l__rb_code_str
% detokenize, then undo LaTeX escapes of the special characters

  {
    \str_set:Nn \l__rb_code_str {#1}
    \str_replace_all:Nnn \l__rb_code_str { \_ } { _ }
    \exp_args:NNnV \str_replace_all:Nnn \l__rb_code_str { \# } \c_hash_str
    \exp_args:NNnV \str_replace_all:Nnn \l__rb_code_str { \% } \c_percent_str
    \exp_args:NNnV \str_replace_all:Nnn \l__rb_code_str { \& } \c_ampersand_str
    \exp_args:NNnV \str_replace_all:Nnn \l__rb_code_str { \$ } \c_dollar_str
    \exp_args:NNnV \str_replace_all:Nnn \l__rb_code_str { \{ } \c_left_brace_str
    \exp_args:NNnV \str_replace_all:Nnn \l__rb_code_str { \} } \c_right_brace_str
  }
\cs_new_protected:Npn \rb@code #1
  { \__rb_clean:n {#1} \exp_args:NV \rb@codeurl \l__rb_code_str }
\cs_new_protected:Npn \rb@acckey #1

% ------------------------------------------------------------------------- section inputs
% \rbinput{candidates}{fallback title}{fallback label}: \input the first file
% sections/<candidate>.tex that exists (candidates are a comma-separated list without spaces);
% otherwise print a placeholder section with the fallback title and label, so cross-references
% to it still resolve. Each choice is written to the log ("rai-bvar: input ..." or
% "rai-bvar: MISSING ..."), which `make check` reads.
\newcommand{\rb@found}{}
\newcommand{\rbinput}[3]{%
  \def\rb@found{}%
  \@for\rb@cand:=#1\do{%
    \ifx\rb@found\@empty
      \IfFileExists{sections/\rb@cand.tex}{\xdef\rb@found{\rb@cand}}{}%
    \fi}%
  \ifx\rb@found\@empty
    \expandafter\@firstoftwo
  \else
    \expandafter\@secondoftwo
  \fi
  {\typeout{rai-bvar: MISSING sections/{#1}.tex}%
   \section{#2}\label{#3}%
   \rb@missing{#1}}%
  {\typeout{rai-bvar: input sections/\rb@found.tex}%
   \input{sections/\rb@found}}}
\newcommand{\rb@missing}[1]{%
  \ifdraftmarks
    \begin{center}\fbox{\parbox{0.85\linewidth}{\centering\color{accmark}
      This section has not been written yet (expected one of
      \texttt{\detokenize{#1}} in \texttt{sections/}).}}\end{center}%
  \else
    \PackageError{rai-bvar}{a section file is missing: \detokenize{#1}}{}%
  \fi}
\makeatother
